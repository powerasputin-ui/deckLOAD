   251→   251→  // Rotate a placement around its center, then validate: clamp inside the deck
   252→   252→  // (with boardOffset) and reject if it collides with other pinned/manual items.
   253→   253→  const tryRotatePlacement = (
   254→   254→    current: { x: number; y: number; width: number; length: number },
   255→   255→    others: { x: number; y: number; width: number; length: number }[]
   256→   256→  ): { x: number; y: number; width: number; length: number } | null => {
   257→   257→    // Keep the center fixed so the rotation visually stays in place
   258→   258→    const cx = current.x + current.width / 2
   259→   259→    const cy = current.y + current.length / 2
   260→   260→    const newW = current.length
   261→   261→    const newL = current.width
   262→   262→    let nx = cx - newW / 2
   263→   263→    let ny = cy - newL / 2
   264→   264→    // Clamp inside the usable area (boardOffset)
   265→   265→    const off = deck.boardOffset
   266→   266→    nx = Math.max(off, Math.min(deck.width - off - newW, nx))
   267→   267→    ny = Math.max(off, Math.min(deck.length - off - newL, ny))
   268→   268→    const clamped = clampToDeck(
   269→   269→      { x: nx, y: ny, width: newW, length: newL },
   270→   270→      deck.width,
   271→   271→      deck.length,
   272→   272→      off
   273→   273→    )
   274→   274→    if (collidesWith({ ...clamped, width: newW, length: newL }, others, deck.gap)) {
   275→   275→      return null
   276→   276→    }
   277→   277→    return { x: clamped.x, y: clamped.y, width: newW, length: newL }
   278→   278→  }

   320→   320→  // Check whether a layer change is allowed for a placement.
   321→   321→  // - maxPhys: physical ceiling from clearance / item.height
   322→   322→  // - totalPlacedForItem: current sum of layers across all placements of this item
   323→   323→  // - item.quantity: total units requested
   324→   324→  // Returns { ok: boolean, reason?: string }
   325→   325→  const checkLayerChange = (
   326→   326→    itemId: string,
   327→   327→    currentLayers: number,
   328→   328→    delta: number,
   329→   329→    excludeId?: string
   330→   330→  ): { ok: boolean; reason?: string; maxPhys: number } => {
   331→   331→    const item = items.find((it) => it.id === itemId)
   332→   332→    if (!item) return { ok: false, reason: 'Груз не найден', maxPhys: 1 }
   333→   333→    const maxPhys = maxLayersFor(item, deck.clearance)
   334→   334→    const newLayers = currentLayers + delta
   335→   335→    if (newLayers < 1) return { ok: false, reason: 'Минимум 1 ярус', maxPhys }
   336→   336→    if (newLayers > maxPhys) {
   337→   337→      return {
   338→   338→        ok: false,
   339→   339→        reason: `Превышена высота под палубой — увеличьте зазор (clearance) в настройках, чтобы добавить ярус`,
   340→   340→        maxPhys,
   341→   341→      }
   342→   342→    }
   343→   343→    // Sum of layers across all placements of this item (excluding the one being changed)
   344→   344→    const sumPlaced =
   345→   345→      pinnedPlacements
   346→   346→        .filter((p) => p.itemId === itemId && p.id !== excludeId)
   347→   347→        .reduce((s, p) => s + p.layers, 0) +
   348→   348→      manualPlacements
   349→   349→        .filter((m) => m.itemId === itemId && m.id !== excludeId)
   350→   350→        .reduce((s, m) => s + Math.max(1, m.layers), 0)
   351→   351→    if (sumPlaced + newLayers > item.quantity) {
   352→   352→      return {
   353→   353→        ok: false,
   354→   354→        reason: `Все ${item.quantity} ед. этого груза уже размещены — увеличьте количество в списке грузов`,
   355→   355→        maxPhys,
   356→   356→      }
   357→   357→    }
   358→   358→    return { ok: true, maxPhys }
   359→   359→  }

   423→   423→  // Group layer change for multiple selected manual placements
   424→   424→  const handleGroupLayerChangeManual = (delta: number) => {
   425→   425→    const selected = manualPlacements.filter((m) => selectedManualIds.includes(m.id))
   426→   426→    let applied = 0
   427→   427→    let blocked = 0
   428→   428→    for (const mp of selected) {
   429→   429→      const current = Math.max(1, mp.layers)
   430→   430→      const check = checkLayerChange(mp.itemId, current, delta, mp.id)
   431→   431→      if (check.ok) {
   432→   432→        updateManualPlacement(mp.id, { layers: current + delta })
   433→   433→        applied++
   434→   434→      } else {
   435→   435→        blocked++
   436→   436→      }
   437→   437→    }
   438→   438→    if (applied > 0 && blocked === 0) {
   439→   439→      toast.success(`Ярусов изменено: ${applied} стоп(ок)`)
   440→   440→    } else if (blocked > 0 && applied === 0) {
   441→   441→      toast.warning(`Невозможно изменить: потолок достигнут (${blocked} стоп.)`)
   442→   442→    } else if (blocked > 0) {
   443→   443→      toast.info(`Изменено ${applied}, блокировано ${blocked} (потолок)`)
   444→   444→    }
   445→   445→  }
   446→   446→
   447→   447→  // Switch mode while preserving placements:
   448→   448→  //  - auto -> manual: all placed items (pinned + auto-packed) become manual placements
   449→   449→  //  - manual -> auto: all manual placements become pinned, auto-packer keeps their positions
   450→   450→  const handleModeChange = (newMode: 'auto' | 'manual') => {
   451→   451→    if (newMode === mode) return
   452→   452→    if (newMode === 'manual') {
   453→   453→      // Convert current auto-mode result (pinned + auto-packed) into manual placements,
   454→   454→      // preserving the number of stacked tiers per footprint.
   455→   455→      const newManual: ManualPlacement[] = result.placed.map((p) => ({
   456→   456→        id: crypto.randomUUID(),
   457→   457→        itemId: p.itemId,
   458→   458→        name: p.name,
   459→   459→        x: p.x,
   460→   460→        y: p.y,
   461→   461→        width: p.width,
   462→   462→        length: p.length,
   463→   463→        layers: Math.max(1, p.stackedCount),
   464→   464→        rotated: p.rotated,
   465→   465→        color: p.color,
   466→   466→        weight: p.weight,
   467→   467→      }))
   468→   468→      useCalculator.setState({
   469→   469→        mode: 'manual',
   470→   470→        manualPlacements: newManual,
   471→   471→        pinnedPlacements: [],
   472→   472→        selectedPinIds: [],
   473→   473→        activeStampId: items[0]?.id ?? null,
   474→   474→      })
   475→   475→      toast.info('Ручной режим — размещения сохранены')
   476→   476→    } else {
   477→   477→      // Convert manual placements into pinned placements, preserving layers.
   478→   478→      const newPinned = manualPlacements.map((m) => ({
   479→   479→        id: crypto.randomUUID(),
   480→   480→        itemId: m.itemId,
   481→   481→        name: m.name,
   482→   482→        x: m.x,
   483→   483→        y: m.y,
   484→   484→        width: m.width,
   485→   485→        length: m.length,
   486→   486→        layers: Math.max(1, m.layers),
   487→   487→        rotated: m.rotated,
   488→   488→        color: m.color,
   489→   489→        weight: m.weight,
   490→   490→      }))
   491→   491→      useCalculator.setState({
   492→   492→        mode: 'auto',
   493→   493→        pinnedPlacements: newPinned,
   494→   494→        manualPlacements: [],
   495→   495→        selectedPinIds: [],
   496→   496→        activeStampId: null,
   497→   497→      })
   498→   498→      toast.info('Авто-режим — размещения сохранены как закреплённые')
   499→   499→    }
   500→   500→  }
   501→   501→
   502→   502→  // Generate several packing variants so the user can choose the best one.
   503→   503→  // Each click produces a fresh set (new seed) — repeated clicks give different options.
   504→   504→  const handleAutoRedistribute = () => {
   505→   505→    const effectiveItems = globalRotation
   506→   506→      ? items
   507→   507→      : items.map((it) => ({ ...it, allowRotation: false }))
   508→   508→    const newVariants = packDeckVariants(
   509→   509→      deck.width,
   510→   510→      deck.length,
   511→   511→      effectiveItems,
   512→   512→      {
   513→   513→        sortStrategy,
   514→   514→        gap: deck.gap,
   515→   515→        boardOffset: deck.boardOffset,
   516→   516→        clearance: deck.clearance,
   517→   517→      },
   518→   518→      3
   519→   519→    )
   520→   520→    setVariants(newVariants)
   521→   521→    // Apply the best variant immediately
   522→   522→    if (newVariants.length > 0) {
   523→   523→      applyVariant(newVariants[0])
   524→   524→    }
   525→   525→    toast.success(`Сгенерировано вариантов: ${newVariants.length}`)
   526→   526→  }
   527→   527→
   528→   528→  // Apply a chosen variant: in manual mode store as manual placements,
   529→   529→  // in auto mode clear pins so the packer result shows through.
   530→   530→  const applyVariant = (variant: PackVariant) => {
   531→   531→    const s = useCalculator.getState()
   532→   532→    if (s.mode === 'manual') {
   533→   533→      const newManual: ManualPlacement[] = variant.result.placed.map((p) => ({
   534→   534→        id: crypto.randomUUID(),
   535→   535→        itemId: p.itemId,
   536→   536→        name: p.name,
   537→   537→        x: p.x,
   538→   538→        y: p.y,
   539→   539→        width: p.width,
   540→   540→        length: p.length,
   541→   541→        layers: Math.max(1, p.stackedCount),
   542→   542→        rotated: p.rotated,
   543→   543→        color: p.color,
   544→   544→        weight: p.weight,
   545→   545→      }))
   546→   546→      useCalculator.setState({
   547→   547→        manualPlacements: newManual,
   548→   548→        pinnedPlacements: [],
   549→   549→        selectedPinIds: [],
   550→   550→      })
   551→   551→    } else {
   552→   552→      useCalculator.setState({
   553→   553→        pinnedPlacements: [],
   554→   554→        manualPlacements: [],
   555→   555→        selectedPinIds: [],
   556→   556→      })
   557→   557→    }
   558→   558→  }
   559→   559→
   560→   560→  const handleSelectVariant = (variant: PackVariant) => {
   561→   561→    applyVariant(variant)
   562→   562→    toast.info(`Применён: ${variant.label} (${variant.utilizationPct}%)`)
   563→   563→  }
   564→   564→
   565→   565→  return (
   566→   566→    <div className="h-screen flex flex-col bg-muted/30 overflow-hidden">
   567→   567→      {/* Top bar */}
   568→   568→      <header className="shrink-0 border-b bg-background/95 backdrop-blur z-30">
   569→   569→        <div className="flex items-center gap-3 px-4 py-2.5">
   570→   570→          <div className="flex items-center gap-2 min-w-0">
   571→   571→            <h1 className="text-sm font-semibold truncate">
   572→   572→              {activeProject?.name ?? 'DeckLoad'}
   573→   573→            </h1>
   574→   574→            {activeProject && (
   575→   575→              <span className="text-xs text-muted-foreground hidden md:inline">
   576→   576→                · {activeProject.deck.width}×{activeProject.deck.length} {activeProject.deck.unit}
   577→   577→              </span>
   578→   578→            )}
   579→   579→          </div>
   580→   580→
   581→   581→          <div className="ml-auto flex items-center gap-2 sm:gap-3">
   582→   582→            <Badge variant="outline" className="hidden lg:inline-flex">
   583→   583→              <Anchor className="h-3 w-3 mr-1" />
   584→   584→              {result.placed.length}/{result.requestedCount} ед. · {Math.round(result.utilization * 100)}%
   585→   585→            </Badge>
   586→   586→
   587→   587→            {/* Mode toggle */}
   588→   588→            <ToggleGroup
   589→   589→              type="single"
   590→   590→              value={mode}
   591→   591→              onValueChange={(v) => {
   592→   592→                if (v) handleModeChange(v as 'auto' | 'manual')
   593→   593→              }}
   594→   594→              className="rounded-lg border bg-card"
   595→   595→            >
   596→   596→              <ToggleGroupItem value="auto" className="px-2.5 data-[state=on]:bg-primary data-[state=on]:text-primary-foreground">
   597→   597→                <Wand2 className="h-3.5 w-3.5 mr-1" />
   598→   598→                <span className="text-xs font-medium hidden sm:inline">Авто</span>
   599→   599→              </ToggleGroupItem>
   600→   600→              <ToggleGroupItem value="manual" className="px-2.5 data-[state=on]:bg-primary data-[state=on]:text-primary-foreground">
   601→   601→                <MousePointerClick className="h-3.5 w-3.5 mr-1" />
   602→   602→                <span className="text-xs font-medium hidden sm:inline">Ручной</span>
   603→   603→              </ToggleGroupItem>
   604→   604→            </ToggleGroup>
   605→   605→          </div>
   606→   606→        </div>
   607→   607→      </header>
   608→   608→
   609→   609→      {/* Workspace */}
   610→   610→      <div className="flex-1 flex min-h-0">
   611→   611→        <Sidebar
   612→   612→          collapsed={sidebarCollapsed}
   613→   613→          onToggle={() => setSidebarCollapsed((v) => !v)}
   614→   614→          onNewCalculation={handleNewCalculation}
   615→   615→          onResetCurrent={handleResetCurrent}
   616→   616→        />
   617→   617→
   618→   618→        {/* Main content */}
   619→   619→        <main className="flex-1 min-w-0 overflow-auto">
   620→   620→          <div className="grid grid-cols-1 xl:grid-cols-12 gap-4 p-4">
   621→   621→            {/* Visualization */}
   622→   622→            <div className="xl:col-span-8">
   623→   623→              <Card className="h-full">
   624→   624→                <CardHeader className="pb-2">
   625→   625→                  <div className="flex items-center justify-between gap-2 flex-wrap">
   626→   626→                    <div>
   627→   627→                      <CardTitle className="text-base flex items-center gap-2">
   628→   628→                        <Ship className="h-4 w-4 text-primary" />
   629→   629→                        Схема палубы
   630→   630→                        {mode === 'manual' && (
   631→   631→                          <Badge variant="secondary" className="text-[10px]">ручной режим</Badge>
   632→   632→                        )}
   633→   633→                      </CardTitle>
   634→   634→                      <CardDescription className="mt-0.5">
   635→   635→                        Вид сверху · зелёная штриховка — свободное пространство
   636→   636→                        {deck.gap > 0 && ` · отступ ${deck.gap} ${UNIT_LABEL[deck.unit]}`}
   637→   637→                      </CardDescription>
   638→   638→                    </div>
   639→   639→                    <div className="flex items-center gap-2 text-xs flex-wrap">
   640→   640→                      <LegendDot color="#0ea5e9" label="Груз" />
   641→   641→                      <LegendDot hatch label="Свободно" />
   642→   642→                      <LegendDot icon="↻" label="Повернут" />
   643→   643→                      {mode === 'auto' && (
   644→   644→                        <LegendDot color="#7c3aed" label="Закреплён" />
   645→   645→                      )}
   646→   646→                    </div>
   647→   647→                  </div>
   648→   648→                </CardHeader>
   649→   649→                <CardContent>
   650→   650→                  <DeckVisualization
   651→   651→                    result={result}
   652→   652→                    unit={deck.unit}
   653→   653→                    gap={deck.gap}
   654→   654→                    boardOffset={deck.boardOffset}
   655→   655→                    showFreeSpace={showFreeSpace}
   656→   656→                    showGrid={showGrid}
   657→   657→                    showLabels={showLabels}
   658→   658→                    hoveredItemId={hoveredItemId}
   659→   659→                    onHover={setHoveredItemId}
   660→   660→                    mode={mode}
   661→   661→                    activeStamp={activeStamp}
   662→   662→                    stampRotated={stampRotated}
   663→   663→                    onPlace={(p) => {
   664→   664→                      const totalRequested = items.reduce((s, it) => s + it.quantity, 0)
   665→   665→                      const placedUnits = manualPlacements.reduce(
   666→   666→                        (s, m) => s + Math.max(1, m.layers),
   667→   667→                        0
   668→   668→                      )
   669→   669→                      if (placedUnits >= totalRequested) {
   670→   670→                        toast.warning('Все грузы уже размещены')
   671→   671→                        return
   672→   672→                      }
   673→   673→                      useCalculator.getState().addManualPlacement(p)
   674→   674→                    }}
   675→   675→                    onMoveManual={(id, x, y) => updateManualPlacement(id, { x, y })}
   676→   676→                    onRemoveManual={removeManualPlacement}
   677→   677→                    manualPlacements={manualPlacements}
   678→   678→                    pinnedPlacements={pinnedPlacements}
   679→   679→                    selectedPinIds={selectedPinIds}
   680→   680→                    onPinPlaced={(p) => pinFromPlaced(p)}
   681→   681→                    onUpdatePinned={(id, x, y) => updatePinned(id, { x, y })}
   682→   682→                    onRemovePinned={handleRemovePinned}
   683→   683→                    onRotatePinned={handleRotatePinned}
   684→   684→                    onTogglePinSelection={togglePinSelection}
   685→   685→                    onClearSelection={clearSelection}
   686→   686→                    onRotateManual={handleRotateManual}
   687→   687→                    onLayerChangePinned={handleLayerChangePinned}
   688→   688→                    onLayerChangeManual={handleLayerChangeManual}
   689→   689→                    getLayerInfo={(itemId, currentLayers, excludeId) => {
   690→   690→                      const item = items.find((it) => it.id === itemId)
   691→   691→                      const maxPhys = item ? maxLayersFor(item, deck.clearance) : 1
   692→   692→                      const incCheck = checkLayerChange(itemId, currentLayers, 1, excludeId)
   693→   693→                      return {
   694→   694→                        maxPhys,
   695→   695→                        canIncrease: incCheck.ok,
   696→   696→                        canDecrease: currentLayers > 1,
   697→   697→                        blockReason: incCheck.ok ? undefined : incCheck.reason,
   698→   698→                      }
   699→   699→                    }}
   700→   700→                    selectedManualIds={selectedManualIds}
   701→   701→                    onToggleManualSelection={toggleManualSelection}
   702→   702→                    onClearManualSelection={clearManualSelection}
   703→   703→                  />
   704→   704→                  <div className="mt-3 flex items-center justify-between text-xs text-muted-foreground">
   705→   705→                    <span>
   706→   706→                      Размещено {result.placed.length} из {result.requestedCount} ед.
   707→   707→                    </span>
   708→   708→                    <span>
   709→   709→                      Загрузка:{' '}
   710→   710→                      <span className="font-semibold text-foreground">
   711→   711→                        {Math.round(result.utilization * 100)}%
   712→   712→                      </span>
   713→   713→                    </span>
   714→   714→                  </div>
   715→   715→                </CardContent>
   716→   716→              </Card>
   717→   717→            </div>
   718→   718→
   719→   719→            {/* Right panel: placement + stats + items */}
   720→   720→            <div className="xl:col-span-4 space-y-4">
   721→   721→              <PlacementPanel
   722→   722→                mode={mode}
   723→   723→                onAutoRedistribute={handleAutoRedistribute}
   724→   724→                variants={variants}
   725→   725→                onSelectVariant={handleSelectVariant}
   726→   726→                onGroupLayerChange={handleGroupLayerChange}
   727→   727→                onGroupLayerChangeManual={handleGroupLayerChangeManual}
   728→   728→                onRemovePinned={handleRemovePinned}
   729→   729→              />
   730→   730→              <StatsPanel result={result} unit={deck.unit} />
   731→   731→              <ItemList
   732→   732→                result={result}
   733→   733→                unit={deck.unit}
   734→   734→                hoveredItemId={hoveredItemId}
   735→   735→                onHover={setHoveredItemId}
   736→   736→              />
   737→   737→            </div>
   738→   738→          </div>
   739→   739→        </main>
   740→   740→      </div>
   741→   741→
   742→   742→      {/* Footer */}
   743→   743→      <footer className="shrink-0 border-t bg-background py-2 px-4">
   744→   744→        <div className="flex items-center justify-between gap-3 text-xs text-muted-foreground">
   745→   745→          <span className="truncate">
   746→   746→            DeckLoad · Maximal Rectangles (BSSF) · вращение, отступы, мульти-проекты
   747→   747→          </span>
   748→   748→          <Button variant="ghost" size="sm" asChild className="h-6 hidden sm:inline-flex">
   749→   749→            <a href="https://github.com" target="_blank" rel="noreferrer" className="text-xs">
   750→   750→              <Github className="h-3 w-3 mr-1" />
   751→   751→              Исходники
   752→   752→            </a>
   753→   753→          </Button>
   754→   754→        </div>
   755→   755→      </footer>
   756→   756→    </div>
   757→   757→  )
   758→   758→}
