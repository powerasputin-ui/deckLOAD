   251→  // Rotate a placement around its center, then validate: clamp inside the deck
   252→  // (with boardOffset) and reject if it collides with other pinned/manual items.
   253→  const tryRotatePlacement = (
   254→    current: { x: number; y: number; width: number; length: number },
   255→    others: { x: number; y: number; width: number; length: number }[]
   256→  ): { x: number; y: number; width: number; length: number } | null => {
   257→    // Keep the center fixed so the rotation visually stays in place
   258→    const cx = current.x + current.width / 2
   259→    const cy = current.y + current.length / 2
   260→    const newW = current.length
   261→    const newL = current.width
   262→    let nx = cx - newW / 2
   263→    let ny = cy - newL / 2
   264→    // Clamp inside the usable area (boardOffset)
   265→    const off = deck.boardOffset
   266→    nx = Math.max(off, Math.min(deck.width - off - newW, nx))
   267→    ny = Math.max(off, Math.min(deck.length - off - newL, ny))
   268→    const clamped = clampToDeck(
   269→      { x: nx, y: ny, width: newW, length: newL },
   270→      deck.width,
   271→      deck.length,
   272→      off
   273→    )
   274→    if (collidesWith({ ...clamped, width: newW, length: newL }, others, deck.gap)) {
   275→      return null
   276→    }
   277→    return { x: clamped.x, y: clamped.y, width: newW, length: newL }
   278→  }

   320→  // Check whether a layer change is allowed for a placement.
   321→  // - maxPhys: physical ceiling from clearance / item.height
   322→  // - totalPlacedForItem: current sum of layers across all placements of this item
   323→  // - item.quantity: total units requested
   324→  // Returns { ok: boolean, reason?: string }
   325→  const checkLayerChange = (
   326→    itemId: string,
   327→    currentLayers: number,
   328→    delta: number,
   329→    excludeId?: string
   330→  ): { ok: boolean; reason?: string; maxPhys: number } => {
   331→    const item = items.find((it) => it.id === itemId)
   332→    if (!item) return { ok: false, reason: 'Груз не найден', maxPhys: 1 }
   333→    const maxPhys = maxLayersFor(item, deck.clearance)
   334→    const newLayers = currentLayers + delta
   335→    if (newLayers < 1) return { ok: false, reason: 'Минимум 1 ярус', maxPhys }
   336→    if (newLayers > maxPhys) {
   337→      return {
   338→        ok: false,
   339→        reason: `Превышена высота под палубой — увеличьте зазор (clearance) в настройках, чтобы добавить ярус`,
   340→        maxPhys,
   341→      }
   342→    }
   343→    // Sum of layers across all placements of this item (excluding the one being changed)
   344→    const sumPlaced =
   345→      pinnedPlacements
   346→        .filter((p) => p.itemId === itemId && p.id !== excludeId)
   347→        .reduce((s, p) => s + p.layers, 0) +
   348→      manualPlacements
   349→        .filter((m) => m.itemId === itemId && m.id !== excludeId)
   350→        .reduce((s, m) => s + Math.max(1, m.layers), 0)
   351→    if (sumPlaced + newLayers > item.quantity) {
   352→      return {
   353→        ok: false,
   354→        reason: `Все ${item.quantity} ед. этого груза уже размещены — увеличьте количество в списке грузов`,
   355→        maxPhys,
   356→      }
   357→    }
   358→    return { ok: true, maxPhys }
   359→  }

   504→  const handleAutoRedistribute = () => {
   505→    const effectiveItems = globalRotation
   506→      ? items
   507→      : items.map((it) => ({ ...it, allowRotation: false }))
   508→    const newVariants = packDeckVariants(
   509→      deck.width,
   510→      deck.length,
   511→      effectiveItems,
   512→      {
   513→        sortStrategy,
   514→        gap: deck.gap,
   515→        boardOffset: deck.boardOffset,
   516→        clearance: deck.clearance,
   517→      },
   518→      3
   519→    )
   520→    setVariants(newVariants)
   521→    // Apply the best variant immediately
   522→    if (newVariants.length > 0) {
   523→      applyVariant(newVariants[0])
   524→    }
   525→    toast.success(`Сгенерировано вариантов: ${newVariants.length}`)
   526→  }
   527→
   528→  // Apply a chosen variant: in manual mode store as manual placements,
   529→  // in auto mode clear pins so the packer result shows through.
   530→  const applyVariant = (variant: PackVariant) => {
   531→    const s = useCalculator.getState()
   532→    if (s.mode === 'manual') {
   533→      const newManual: ManualPlacement[] = variant.result.placed.map((p) => ({
   534→        id: crypto.randomUUID(),
   535→        itemId: p.itemId,
   536→        name: p.name,
   537→        x: p.x,
   538→        y: p.y,
   539→        width: p.width,
   540→        length: p.length,
   541→        layers: Math.max(1, p.stackedCount),
   542→        rotated: p.rotated,
   543→        color: p.color,
   544→        weight: p.weight,
   545→      }))
   546→      useCalculator.setState({
   547→        manualPlacements: newManual,
   548→        pinnedPlacements: [],
   549→        selectedPinIds: [],
   550→      })
   551→    } else {
   552→      useCalculator.setState({
   553→        pinnedPlacements: [],
   554→        manualPlacements: [],
   555→        selectedPinIds: [],
   556→      })
   557→    }
   558→  }
   559→
   560→  const handleSelectVariant = (variant: PackVariant) => {
   561→    applyVariant(variant)
   562→    toast.info(`Применён: ${variant.label} (${variant.utilizationPct}%)`)
   563→  }
   564→
   565→  return (
   566→    <div className="h-screen flex flex-col bg-muted/30 overflow-hidden">
   567→      {/* Top bar */}
   568→      <header className="shrink-0 border-b bg-background/95 backdrop-blur z-30">
   569→        <div className="flex items-center gap-3 px-4 py-2.5">
   570→          <div className="flex items-center gap-2 min-w-0">
   571→            <h1 className="text-sm font-semibold truncate">
   572→              {activeProject?.name ?? 'DeckLoad'}
   573→            </h1>
   574→            {activeProject && (
   575→              <span className="text-xs text-muted-foreground hidden md:inline">
   576→                · {activeProject.deck.width}×{activeProject.deck.length} {activeProject.deck.unit}
   577→              </span>
   578→            )}
   579→          </div>
   580→
   581→          <div className="ml-auto flex items-center gap-2 sm:gap-3">
   582→            <Badge variant="outline" className="hidden lg:inline-flex">
   583→              <Anchor className="h-3 w-3 mr-1" />
   584→              {result.placed.length}/{result.requestedCount} ед. · {Math.round(result.utilization * 100)}%
   585→            </Badge>
   586→
   587→            {/* Mode toggle */}
   588→            <ToggleGroup
   589→              type="single"
   590→              value={mode}
   591→              onValueChange={(v) => {
   592→                if (v) handleModeChange(v as 'auto' | 'manual')
   593→              }}
   594→              className="rounded-lg border bg-card"
   595→            >
   596→              <ToggleGroupItem value="auto" className="px-2.5 data-[state=on]:bg-primary data-[state=on]:text-primary-foreground">
   597→                <Wand2 className="h-3.5 w-3.5 mr-1" />
   598→                <span className="text-xs font-medium hidden sm:inline">Авто</span>
   599→              </ToggleGroupItem>
   600→              <ToggleGroupItem value="manual" className="px-2.5 data-[state=on]:bg-primary data-[state=on]:text-primary-foreground">
   601→                <MousePointerClick className="h-3.5 w-3.5 mr-1" />
   602→                <span className="text-xs font-medium hidden sm:inline">Ручной</span>
   603→              </ToggleGroupItem>
   604→            </ToggleGroup>
   605→          </div>
   606→        </div>
   607→      </header>
   608→
   609→      {/* Workspace */}
   610→      <div className="flex-1 flex min-h-0">
   611→        <Sidebar
   612→          collapsed={sidebarCollapsed}
   613→          onToggle={() => setSidebarCollapsed((v) => !v)}
   614→          onNewCalculation={handleNewCalculation}
   615→          onResetCurrent={handleResetCurrent}
   616→        />
   617→
   618→        {/* Main content */}
   619→        <main className="flex-1 min-w-0 overflow-auto">
   620→          <div className="grid grid-cols-1 xl:grid-cols-12 gap-4 p-4">
   621→            {/* Visualization */}
   622→            <div className="xl:col-span-8">
   623→              <Card className="h-full">
   624→                <CardHeader className="pb-2">
   625→                  <div className="flex items-center justify-between gap-2 flex-wrap">
   626→                    <div>
   627→                      <CardTitle className="text-base flex items-center gap-2">
   628→                        <Ship className="h-4 w-4 text-primary" />
   629→                        Схема палубы
   630→                        {mode === 'manual' && (
   631→                          <Badge variant="secondary" className="text-[10px]">ручной режим</Badge>
   632→                        )}
   633→                      </CardTitle>
   634→                      <CardDescription className="mt-0.5">
   635→                        Вид сверху · зелёная штриховка — свободное пространство
   636→                        {deck.gap > 0 && ` · отступ ${deck.gap} ${UNIT_LABEL[deck.unit]}`}
   637→                      </CardDescription>
   638→                    </div>
   639→                    <div className="flex items-center gap-2 text-xs flex-wrap">
   640→                      <LegendDot color="#0ea5e9" label="Груз" />
   641→                      <LegendDot hatch label="Свободно" />
   642→                      <LegendDot icon="↻" label="Повернут" />
   643→                      {mode === 'auto' && (
   644→                        <LegendDot color="#7c3aed" label="Закреплён" />
   645→                      )}
   646→                    </div>
   647→                  </div>
   648→                </CardHeader>
   649→                <CardContent>
   650→                  <DeckVisualization
   651→                    result={result}
   652→                    unit={deck.unit}
   653→                    gap={deck.gap}
   654→                    boardOffset={deck.boardOffset}
   655→                    showFreeSpace={showFreeSpace}
   656→                    showGrid={showGrid}
   657→                    showLabels={showLabels}
   658→                    hoveredItemId={hoveredItemId}
   659→                    onHover={setHoveredItemId}
   660→                    mode={mode}
   661→                    activeStamp={activeStamp}
   662→                    stampRotated={stampRotated}
   663→                    onPlace={(p) => {
   664→                      const totalRequested = items.reduce((s, it) => s + it.quantity, 0)
   665→                      const placedUnits = manualPlacements.reduce(
   666→                        (s, m) => s + Math.max(1, m.layers),
   667→                        0
   668→                      )
   669→                      if (placedUnits >= totalRequested) {
   670→                        toast.warning('Все грузы уже размещены')
   671→                        return
   672→                      }
   673→                      useCalculator.getState().addManualPlacement(p)
   674→                    }}
   675→                    onMoveManual={(id, x, y) => updateManualPlacement(id, { x, y })}
   676→                    onRemoveManual={removeManualPlacement}
   677→                    manualPlacements={manualPlacements}
   678→                    pinnedPlacements={pinnedPlacements}
   679→                    selectedPinIds={selectedPinIds}
   680→                    onPinPlaced={(p) => pinFromPlaced(p)}
   681→                    onUpdatePinned={(id, x, y) => updatePinned(id, { x, y })}
   682→                    onRemovePinned={handleRemovePinned}
   683→                    onRotatePinned={handleRotatePinned}
   684→                    onTogglePinSelection={togglePinSelection}
   685→                    onClearSelection={clearSelection}
   686→                    onRotateManual={handleRotateManual}
   687→                    onLayerChangePinned={handleLayerChangePinned}
   688→                    onLayerChangeManual={handleLayerChangeManual}
   689→                    getLayerInfo={(itemId, currentLayers, excludeId) => {
   690→                      const item = items.find((it) => it.id === itemId)
   691→                      const maxPhys = item ? maxLayersFor(item, deck.clearance) : 1
   692→                      const incCheck = checkLayerChange(itemId, currentLayers, 1, excludeId)
   693→                      return {
   694→                        maxPhys,
   695→                        canIncrease: incCheck.ok,
   696→                        canDecrease: currentLayers > 1,
   697→                        blockReason: incCheck.ok ? undefined : incCheck.reason,
   698→                      }
   699→                    }}
   700→                    selectedManualIds={selectedManualIds}
   701→                    onToggleManualSelection={toggleManualSelection}
   702→                    onClearManualSelection={clearManualSelection}
   703→                  />
   704→                  <div className="mt-3 flex items-center justify-between text-xs text-muted-foreground">
   705→                    <span>
   706→                      Размещено {result.placed.length} из {result.requestedCount} ед.
   707→                    </span>
   708→                    <span>
   709→                      Загрузка:{' '}
   710→                      <span className="font-semibold text-foreground">
   711→                        {Math.round(result.utilization * 100)}%
   712→                      </span>
   713→                    </span>
   714→                  </div>
   715→                </CardContent>
   716→              </Card>
   717→            </div>
   718→
   719→            {/* Right panel: placement + stats + items */}
   720→            <div className="xl:col-span-4 space-y-4">
   721→              <PlacementPanel
   722→                mode={mode}
   723→                onAutoRedistribute={handleAutoRedistribute}
   724→                variants={variants}
   725→                onSelectVariant={handleSelectVariant}
   726→                onGroupLayerChange={handleGroupLayerChange}
   727→                onGroupLayerChangeManual={handleGroupLayerChangeManual}
   728→                onRemovePinned={handleRemovePinned}
   729→              />
   730→              <StatsPanel result={result} unit={deck.unit} />
   731→              <ItemList
   732→                result={result}
   733→                unit={deck.unit}
   734→                hoveredItemId={hoveredItemId}
   735→                onHover={setHoveredItemId}
   736→              />
   737→            </div>
   738→          </div>
   739→        </main>
   740→      </div>
   741→
   742→      {/* Footer */}
   743→      <footer className="shrink-0 border-t bg-background py-2 px-4">
   744→        <div className="flex items-center justify-between gap-3 text-xs text-muted-foreground">
   745→          <span className="truncate">
   746→            DeckLoad · Maximal Rectangles (BSSF) · вращение, отступы, мульти-проекты
   747→          </span>
   748→          <Button variant="ghost" size="sm" asChild className="h-6 hidden sm:inline-flex">
   749→            <a href="https://github.com" target="_blank" rel="noreferrer" className="text-xs">
   750→              <Github className="h-3 w-3 mr-1" />
   751→              Исходники
   752→            </a>
   753→          </Button>
   754→        </div>
   755→      </footer>
   756→    </div>
   757→  )
   758→}

   764→  icon,
   765→}: {
   766→  color?: string
   767→  label: string
   768→  hatch?: boolean
   769→  icon?: string

   776→          style={{

   778→              'repeating-linear-gradient(45deg, rgba(16,185,129,0.35) 0 2px, transparent 2px 5px)',
   779→          }}
   780→        />

   786→        <span
   787→          className="h-3 w-3 rounded-sm border border-black/10"
   788→          style={{ backgroundColor: color }}
   789→        />
   790→      )}
   791→      {label}
   792→    </span>
   793→  )
