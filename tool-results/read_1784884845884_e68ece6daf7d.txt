     1→     1→'use client'
     2→     2→
     3→     3→import { useMemo, useRef, useState, useCallback } from 'react'
     4→     4→import {
     5→     5→  computeFreeRects,
     6→     6→  clampToDeck,
     7→     7→  collidesWith,
     8→     8→  type PackingResult,
     9→     9→  type PlacedItem,
    10→    10→  type ManualPlacement,
    11→    11→  type PinnedPlacement,
    12→    12→} from '@/lib/packing'
    13→    13→import { UNIT_LABEL } from '@/store/calculator'
    14→    14→import { v4 as uuid } from 'uuid'
    15→    15→import { toast } from 'sonner'
    16→    16→
    17→    17→interface DeckVisualizationProps {
    18→    18→  result: PackingResult
    19→    19→  unit: 'm' | 'cm' | 'ft'
    20→    20→  gap: number
    21→    21→  boardOffset: number
    22→    22→  showFreeSpace: boolean
    23→    23→  showGrid: boolean
    24→    24→  showLabels: boolean
    25→    25→  hoveredItemId: string | null
    26→    26→  onHover: (id: string | null) => void
    27→    27→  mode: 'auto' | 'manual'
    28→    28→  activeStamp: { id: string; width: number; length: number; color: string; name: string; weight?: number } | null
    29→    29→  stampRotated: boolean
    30→    30→  onPlace?: (p: ManualPlacement) => void
    31→    31→  onMoveManual?: (id: string, x: number, y: number) => void
    32→    32→  onRemoveManual?: (id: string) => void
    33→    33→  manualPlacements: ManualPlacement[]
    34→    34→  // Interactive auto mode
    35→    35→  pinnedPlacements: PinnedPlacement[]
    36→    36→  selectedPinIds: string[]
    37→    37→  onPinPlaced?: (placed: { itemId: string; name: string; x: number; y: number; width: number; length: number; layers: number; rotated: boolean; color: string; weight?: number }) => string | undefined
    38→    38→  onUpdatePinned?: (id: string, x: number, y: number) => void
    39→    39→  onRemovePinned?: (id: string) => void
    40→    40→  onRotatePinned?: (id: string) => void
    41→    41→  onTogglePinSelection?: (id: string, additive: boolean) => void
    42→    42→  onClearSelection?: () => void
    43→    43→  // Manual mode rotate
    44→    44→  onRotateManual?: (id: string) => void
    45→    45→  // Layer change (per-placement)
    46→    46→  onLayerChangePinned?: (id: string, delta: number) => void
    47→    47→  onLayerChangeManual?: (id: string, delta: number) => void
    48→    48→  // Layer validation info for single-selected placement
    49→    49→  getLayerInfo?: (itemId: string, currentLayers: number, excludeId?: string) => { maxPhys: number; canIncrease: boolean; canDecrease: boolean; blockReason?: string }
    50→    50→  // Manual multi-selection
    51→    51→  selectedManualIds?: string[]
    52→    52→  onToggleManualSelection?: (id: string, additive: boolean) => void
    53→    53→  onClearManualSelection?: () => void
    54→    54→}
    55→    55→
    56→    56→export function DeckVisualization({
    57→    57→  result,
    58→    58→  unit,
    59→    59→  gap,
    60→    60→  boardOffset,
    61→    61→  showFreeSpace,
    62→    62→  showGrid,
    63→    63→  showLabels,
    64→    64→  hoveredItemId,
    65→    65→  onHover,
    66→    66→  mode,
    67→    67→  activeStamp,
    68→    68→  stampRotated,
    69→    69→  onPlace,
    70→    70→  onMoveManual,
    71→    71→  onRemoveManual,
    72→    72→  manualPlacements,
    73→    73→  pinnedPlacements,
    74→    74→  selectedPinIds,
    75→    75→  onPinPlaced,
    76→    76→  onUpdatePinned,
    77→    77→  onRemovePinned,
    78→    78→  onRotatePinned,
    79→    79→  onTogglePinSelection,
    80→    80→  onClearSelection,
    81→    81→  onRotateManual,
    82→    82→  onLayerChangePinned,
    83→    83→  onLayerChangeManual,
    84→    84→  getLayerInfo,
    85→    85→  selectedManualIds,
    86→    86→  onToggleManualSelection,
    87→    87→  onClearManualSelection,
    88→    88→}: DeckVisualizationProps) {
    89→    89→  const { deckWidth, deckLength } = result
    90→    90→  const svgRef = useRef<SVGSVGElement>(null)
    91→    91→  const [hoverPos, setHoverPos] = useState<{ x: number; y: number } | null>(null)
    92→    92→  const [dragState, setDragState] = useState<{
    93→    93→    id: string
    94→    94→    startMouse: { x: number; y: number }
    95→    95→    startPlace: { x: number; y: number }
    96→    96→  } | null>(null)
    97→    97→  const [pinDrag, setPinDrag] = useState<{
    98→    98→    id: string
    99→    99→    startDeck: { x: number; y: number }
   100→   100→    startPlace: { x: number; y: number }
   101→   101→    moved: boolean
   102→   102→  } | null>(null)
   103→   103→  const [selectedManual, setSelectedManual] = useState<string | null>(null)
   104→   104→
   105→   105→  const isInteractiveAuto = mode === 'auto' && onPinPlaced && onUpdatePinned
   106→   106→
   107→   107→  const edgePad = boardOffset
   108→   108→
   109→   109→  const freeRects = useMemo(
   110→   110→    () => computeFreeRects(deckWidth, deckLength, result.placed, gap, boardOffset),
   111→   111→    [deckWidth, deckLength, result.placed, gap, boardOffset]
   112→   112→  )
   113→   113→
   114→   114→  // Layout geometry
   115→   115→  const maxW = 900
   116→   116→  const maxH = 560
   117→   117→  const pad = 32
   118→   118→  const scale = Math.min(
   119→   119→    (maxW - pad * 2) / Math.max(deckWidth, 1),
   120→   120→    (maxH - pad * 2) / Math.max(deckLength, 1)
   121→   121→  )
   122→   122→  const w = deckWidth * scale
   123→   123→  const h = deckLength * scale
   124→   124→  const offX = (maxW - w) / 2
   125→   125→  const offY = (maxH - h) / 2
   126→   126→
   127→   127→  const gridStep = (() => {
   128→   128→    const dim = Math.max(deckWidth, deckLength)
   129→   129→    if (dim <= 6) return 0.5
   130→   130→    if (dim <= 20) return 1
   131→   131→    if (dim <= 60) return 5
   132→   132→    return 10
   133→   133→  })()
   134→   134→
   135→   135→  // Convert screen px (inside deck) -> deck coords
   136→   136→  const screenToDeck = useCallback(
   137→   137→    (clientX: number, clientY: number) => {
   138→   138→      const svg = svgRef.current
   139→   139→      if (!svg) return null
   140→   140→      const pt = svg.createSVGPoint()
   141→   141→      pt.x = clientX
   142→   142→      pt.y = clientY
   143→   143→      const ctm = svg.getScreenCTM()
   144→   144→      if (!ctm) return null
   145→   145→      const p = pt.matrixTransform(ctm.inverse())
   146→   146→      const dx = (p.x - offX) / scale
   147→   147→      const dy = (p.y - offY) / scale
   148→   148→      return { x: dx, y: dy }
   149→   149→    },
   150→   150→    [offX, offY, scale]
   151→   151→  )
   152→   152→
   153→   153→  const toX = (v: number) => offX + v * scale
   154→   154→  const toY = (v: number) => offY + v * scale
   155→   155→  const fmt = (v: number) => {
   156→   156→    const r = Math.round(v * 100) / 100
   157→   157→    return Number.isInteger(r) ? `${r}` : r.toFixed(2)
   158→   158→  }
   159→   159→
   160→   160→  const stampDims = activeStamp
   161→   161→    ? stampRotated
   162→   162→      ? { w: activeStamp.length, l: activeStamp.width }
   163→   163→      : { w: activeStamp.width, l: activeStamp.length }
   164→   164→    : null
   165→   165→
   166→   166→  const handleDeckClick = (e: React.MouseEvent) => {
   167→   167→    if (mode !== 'manual' || !activeStamp || !stampDims || !onPlace) return
   168→   168→    const pos = screenToDeck(e.clientX, e.clientY)
   169→   169→    if (!pos) return
   170→   170→    // Snap so the item's top-left is at the cursor, then clamp inside deck
   171→   171→    const clamped = clampToDeck(
   172→   172→      { x: pos.x, y: pos.y, width: stampDims.w, length: stampDims.l },
   173→   173→      deckWidth,
   174→   174→      deckLength,
   175→   175→      edgePad
   176→   176→    )
   177→   177→    // Check collision with existing manual placements
   178→   178→    const others = manualPlacements
   179→   179→      .filter((m) => m.id !== 'preview')
   180→   180→      .map((m) => ({ x: m.x, y: m.y, width: m.width, length: m.length }))
   181→   181→    if (collidesWith({ ...clamped, width: stampDims.w, length: stampDims.l }, others, gap)) {
   182→   182→      return // ignore overlapping placement
   183→   183→    }
   184→   184→    onPlace({
   185→   185→      id: uuid(),
   186→   186→      itemId: activeStamp.id,
   187→   187→      name: activeStamp.name,
   188→   188→      x: clamped.x,
   189→   189→      y: clamped.y,
   190→   190→      width: stampDims.w,
   191→   191→      length: stampDims.l,
   192→   192→      layers: 1,
   193→   193→      rotated: stampRotated,
   194→   194→      color: activeStamp.color,
   195→   195→      weight: activeStamp.weight,
   196→   196→    })
   197→   197→  }
   198→   198→
   199→   199→  const handleManualPointerDown = (
   200→   200→    e: React.PointerEvent,
   201→   201→    mp: ManualPlacement
   202→   202→  ) => {
   203→   203→    if (mode !== 'manual') return
   204→   204→    e.stopPropagation()
   205→   205→    const additive = e.shiftKey || e.ctrlKey || e.metaKey
   206→   206→    if (additive) {
   207→   207→      // Toggle selection without starting a drag
   208→   208→      onToggleManualSelection?.(mp.id, true)
   209→   209→      return
   210→   210→    }
   211→   211→    setSelectedManual(mp.id)
   212→   212→    setDragState({
   213→   213→      id: mp.id,
   214→   214→      startMouse: { x: e.clientX, y: e.clientY },
   215→   215→      startPlace: { x: mp.x, y: mp.y },
   216→   216→    })
   217→   217→    ;(e.target as Element).setPointerCapture?.(e.pointerId)
   218→   218→  }
   219→   219→
   220→   220→  // Pin a placed auto-mode item so it becomes user-controlled
   221→   221→  const handlePinPointerDown = (
   222→   222→    e: React.PointerEvent,
   223→   223→    placed: PlacedItem,
   224→   224→    existingPinId?: string
   225→   225→  ) => {
   226→   226→    if (!isInteractiveAuto) return
   227→   227→    e.stopPropagation()
   228→   228→    const additive = e.shiftKey || e.ctrlKey || e.metaKey
   229→   229→    // If clicking an already-pinned item, keep it; otherwise create a pin
   230→   230→    let pinId = existingPinId
   231→   231→    if (!pinId) {
   232→   232→      pinId = onPinPlaced({
   233→   233→        itemId: placed.itemId,
   234→   234→        name: placed.name,
   235→   235→        x: placed.x,
   236→   236→        y: placed.y,
   237→   237→        width: placed.width,
   238→   238→        length: placed.length,
   239→   239→        layers: placed.layers,
   240→   240→        rotated: placed.rotated,
   241→   241→        color: placed.color,
   242→   242→        weight: placed.weight,
   243→   243→      })
   244→   244→    } else if (!additive) {
   245→   245→      onTogglePinSelection?.(pinId, false)
   246→   246→    }
   247→   247→    if (additive && pinId) {
   248→   248→      onTogglePinSelection?.(pinId, true)
   249→   249→    }
   250→   250→    const startDeck = screenToDeck(e.clientX, e.clientY)
   251→   251→    if (!startDeck || !pinId) return
   252→   252→    setPinDrag({
   253→   253→      id: pinId,
   254→   254→      startDeck,
   255→   255→      startPlace: { x: placed.x, y: placed.y },
   256→   256→      moved: false,
   257→   257→    })
   258→   258→    ;(e.target as Element).setPointerCapture?.(e.pointerId)
   259→   259→  }
   260→   260→
   261→   261→  // Gap-aware collision resolution with "sliding" along obstacles.
   262→   262→  // Tries full delta, then X-only, then Y-only; if all collide, performs a
   263→   263→  // binary search along the movement vector to slide as close to the target as
   264→   264→  // possible without overlapping neighbours (instead of snapping back to start).
   265→   265→  const resolveDragPosition = (
   266→   266→    targetX: number,
   267→   267→    targetY: number,
   268→   268→    width: number,
   269→   269→    length: number,
   270→   270→    currentX: number,
   271→   271→    currentY: number,
   272→   272→    others: { x: number; y: number; width: number; length: number }[]
   273→   273→  ): { x: number; y: number } => {
   274→   274→    const tryPos = (x: number, y: number): { x: number; y: number } | null => {
   275→   275→      const clamped = clampToDeck(
   276→   276→        { x, y, width, length },
   277→   277→        deckWidth,
   278→   278→        deckLength,
   279→   279→        edgePad
   280→   280→      )
   281→   281→      if (!collidesWith({ ...clamped, width, length }, others, gap)) {
   282→   282→        return { x: clamped.x, y: clamped.y }
   283→   283→      }
   284→   284→      return null
   285→   285→    }
   286→   286→
   287→   287→    // 1) full delta, 2) X-only, 3) Y-only
   288→   288→    const candidates: { x: number; y: number }[] = [
   289→   289→      { x: targetX, y: targetY },
   290→   290→      { x: targetX, y: currentY },
   291→   291→      { x: currentX, y: targetY },
   292→   292→    ]
   293→   293→    for (const c of candidates) {
   294→   294→      const res = tryPos(c.x, c.y)
   295→   295→      if (res) return res
   296→   296→    }
   297→   297→
   298→   298→    // 4) Binary search along the movement vector to slide as close as possible
   299→   299→    let lo = 0
   300→   300→    let hi = 1
   301→   301→    let best: { x: number; y: number } | null = null
   302→   302→    for (let i = 0; i < 10; i++) {
   303→   303→      const mid = (lo + hi) / 2
   304→   304→      const x = currentX + (targetX - currentX) * mid
   305→   305→      const y = currentY + (targetY - currentY) * mid
   306→   306→      const res = tryPos(x, y)
   307→   307→      if (res) {
   308→   308→        best = res
   309→   309→        lo = mid
   310→   310→      } else {
   311→   311→        hi = mid
   312→   312→      }
   313→   313→    }
   314→   314→    return best ?? { x: currentX, y: currentY }
   315→   315→  }
   316→   316→
   317→   317→  const handlePointerMove = (e: React.PointerEvent) => {
   318→   318→    if (mode === 'manual' && activeStamp && !dragState) {
   319→   319→      const pos = screenToDeck(e.clientX, e.clientY)
   320→   320→      if (pos) setHoverPos(pos)
   321→   321→    }
   322→   322→    if (dragState && onMoveManual) {
   323→   323→      const pos = screenToDeck(e.clientX, e.clientY)
   324→   324→      if (!pos) return
   325→   325→      const startDeck = screenToDeck(dragState.startMouse.x, dragState.startMouse.y)
   326→   326→      if (!startDeck) return
   327→   327→      const deltaX = pos.x - startDeck.x
   328→   328→      const deltaY = pos.y - startDeck.y
   329→   329→      const nx = dragState.startPlace.x + deltaX
   330→   330→      const ny = dragState.startPlace.y + deltaY
   331→   331→      const mp = manualPlacements.find((m) => m.id === dragState.id)
   332→   332→      if (mp) {
   333→   333→        const others = manualPlacements
   334→   334→          .filter((m) => m.id !== dragState.id)
   335→   335→          .map((m) => ({ x: m.x, y: m.y, width: m.width, length: m.length }))
   336→   336→        const resolved = resolveDragPosition(
   337→   337→          nx, ny, mp.width, mp.length, mp.x, mp.y, others
   338→   338→        )
   339→   339→        onMoveManual(dragState.id, resolved.x, resolved.y)
   340→   340→      }
   341→   341→    }
   342→   342→    if (pinDrag && onUpdatePinned) {
   343→   343→      const pos = screenToDeck(e.clientX, e.clientY)
   344→   344→      if (!pos) return
   345→   345→      const deltaX = pos.x - pinDrag.startDeck.x
   346→   346→      const deltaY = pos.y - pinDrag.startDeck.y
   347→   347→      if (Math.abs(deltaX) > 0.1 || Math.abs(deltaY) > 0.1) {
   348→   348→        setPinDrag((d) => (d ? { ...d, moved: true } : d))
   349→   349→      }
   350→   350→      const pin = pinnedPlacements.find((p) => p.id === pinDrag.id)
   351→   351→      if (!pin) return
   352→   352→      const nx = pinDrag.startPlace.x + deltaX
   353→   353→      const ny = pinDrag.startPlace.y + deltaY
   354→   354→      // Prevent overlap with OTHER pinned items (auto-packed items reflow)
   355→   355→      const others = pinnedPlacements
   356→   356→        .filter((p) => p.id !== pinDrag.id)
   357→   357→        .map((p) => ({ x: p.x, y: p.y, width: p.width, length: p.length }))
   358→   358→      const resolved = resolveDragPosition(
   359→   359→        nx, ny, pin.width, pin.length, pin.x, pin.y, others
   360→   360→      )
   361→   361→      onUpdatePinned(pinDrag.id, resolved.x, resolved.y)
   362→   362→    }
   363→   363→  }
   364→   364→
   365→   365→  const handlePointerUp = (e: React.PointerEvent) => {
   366→   366→    // Click on empty deck area in interactive auto mode clears selection
   367→   367→    if (isInteractiveAuto && !pinDrag && !dragState) {
   368→   368→      const target = e.target as Element
   369→   369→      // Only clear if clicked directly on the SVG background or deck rect
   370→   370→      if (target.tagName === 'rect' || target.tagName === 'svg' || target.tagName === 'SVG') {
   371→   371→        const fill = target.getAttribute('fill')
   372→   372→        if (fill === '#ffffff' || fill === 'url(#deck-grid)' || target.tagName === 'svg') {
   373→   373→          onClearSelection?.()
   374→   374→        }
   375→   375→      }
   376→   376→    }
   377→   377→    setDragState(null)
   378→   378→    setPinDrag(null)
   379→   379→  }
   380→   380→
   381→   381→  const handleManualLeave = () => {
   382→   382→    setHoverPos(null)
   383→   383→  }
   384→   384→
   385→   385→  const hasContent = result.placed.length > 0 || deckWidth > 0
   386→   386→
   387→   387→  // Manual mode: rendered items come from manualPlacements; auto: from result.placed
   388→   388→  const renderedItems: (PlacedItem & { manualId?: string })[] =
   389→   389→    mode === 'manual'
   390→   390→      ? manualPlacements.map((m, i) => ({
   391→   391→          itemId: m.itemId,
   392→   392→          name: m.name,
   393→   393→          x: m.x,
   394→   394→          y: m.y,
   395→   395→          width: m.width,
   396→   396→          length: m.length,
   397→   397→          height: 0,
   398→   398→          layers: Math.max(1, m.layers),
   399→   399→          stackedCount: Math.max(1, m.layers),
   400→   400→          rotated: m.rotated,
   401→   401→          color: m.color,
   402→   402→          weight: m.weight,
   403→   403→          index: i,
   404→   404→          manualId: m.id,
   405→   405→        }))
   406→   406→      : result.placed
   407→   407→
   408→   408→  return (
   409→   409→    <div className="w-full overflow-x-auto" onMouseLeave={handleManualLeave}>
   410→   410→      <svg
   411→   411→        ref={svgRef}
   412→   412→        viewBox={`0 0 ${maxW} ${maxH}`}
   413→   413→        className="w-full h-auto"
   414→   414→        style={{ maxHeight: 560, cursor: mode === 'manual' && activeStamp ? 'crosshair' : 'default' }}
   415→   415→        onClick={mode === 'manual' ? handleDeckClick : undefined}
   416→   416→        onPointerMove={handlePointerMove}
   417→   417→        onPointerUp={handlePointerUp}
   418→   418→      >
   419→   419→        <defs>
   420→   420→          <pattern
   421→   421→            id="deck-grid"
   422→   422→            width={gridStep * scale}
   423→   423→            height={gridStep * scale}
   424→   424→            patternUnits="userSpaceOnUse"
   425→   425→          >
   426→   426→            <path
   427→   427→              d={`M ${gridStep * scale} 0 L 0 0 0 ${gridStep * scale}`}
   428→   428→              fill="none"
   429→   429→              stroke="#cbd5e1"
   430→   430→              strokeWidth={0.5}
   431→   431→            />
   432→   432→          </pattern>
   433→   433→          <pattern id="free-hatch" width="8" height="8" patternUnits="userSpaceOnUse" patternTransform="rotate(45)">
   434→   434→            <rect width="8" height="8" fill="transparent" />
   435→   435→            <line x1="0" y1="0" x2="0" y2="8" stroke="rgba(16,185,129,0.28)" strokeWidth="3" />
   436→   436→          </pattern>
   437→   437→        </defs>
   438→   438→
   439→   439→        {/* Deck background */}
   440→   440→        <rect
   441→   441→          x={offX}
   442→   442→          y={offY}
   443→   443→          width={w}
   444→   444→          height={h}
   445→   445→          rx={6}
   446→   446→          fill="#ffffff"
   447→   447→          stroke="#1e293b"
   448→   448→          strokeWidth={2}
   449→   449→        />
   450→   450→        {showGrid && hasContent && (
   451→   451→          <rect x={offX} y={offY} width={w} height={h} rx={6} fill="url(#deck-grid)" />
   452→   452→        )}
   453→   453→
   454→   454→        {/* Edge padding border (usable region) */}
   455→   455→        {edgePad > 0 && (
   456→   456→          <rect
   457→   457→            x={toX(edgePad)}
   458→   458→            y={toY(edgePad)}
   459→   459→            width={(deckWidth - edgePad * 2) * scale}
   460→   460→            height={(deckLength - edgePad * 2) * scale}
   461→   461→            fill="none"
   462→   462→            stroke="#94a3b8"
   463→   463→            strokeWidth={0.75}
   464→   464→            strokeDasharray="3 3"

   764→   764→function PlacedRect({
